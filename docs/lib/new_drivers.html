106    def _checksum(self, instr_code, data):
107        """
108        Calcola il checksum per un messaggio, basato sul codice di istruzione e sui dati.
109        Il checksum è il complemento a 1 della somma di tutti i byte del messaggio (escluso l'Header), modulo 256.
110        
111        Args:
112            instr_code (bytes): Il codice di istruzione del messaggio.
113            data (bytes): I dati del messaggio.
114
115        Returns:
116            bytes: Il checksum calcolato in formato byte.
117
118        """
119
120        lenght = self._data_len(data)
121
122        # il checksum è calcolato su instr_code + lenght + data, escludendo l'Header e l'indirizzo
123        checksum = 255 - (sum(instr_code + lenght + data) % 256)
124
125        return checksum.to_bytes(1, byteorder='little')

127    def _build_message(self, comando, data=None):
128        """
129        Costruisce un messaggio da inviare alla bilancia, basato sul comando e sui dati forniti.
130
131        Args:
132            comando (str): Il comando da inviare, deve essere una chiave presente in self.comandi.
133            data (varia, optional): I dati associati al comando, se richiesti. La forma e il contenuto dei dati dipendono dal comando specifico.
134        
135        Returns:
136            bytes: Il messaggio completo da inviare alla bilancia, in formato byte.
137        """
138
139        # ogni istruzione ha un codice specifico, che è definito in self.comandi. Se il comando non è riconosciuto, solleva un errore.
140        instr_code = self.comandi[comando]
141
142        # alcuni comandi hanno argomenti specifici che devono essere codificati in un certo modo. Qui gestiamo la codifica dei dati in base al comando.
143        if data is not None:
144            # remote activation ha un byte di dati che dipende dall'azione (start, stop, shutter)
145            if comando == "Remote activation":
146                data_encoded = self.remote_activation_data[data]
147            
148            # data-logging ha due byte di dati che rappresentano una configurazione di quali parametri loggare, codificati come bit in due byte
149            elif comando == "Config data-logging":
150
151                byte1 = 0
152                byte2 = 0
153
154                for item in self.config_data_logging_data[0]:
155                    if item in data:
156                        byte1 |= (1 << self.config_data_logging_data[0].index(item))
157
158                for item in self.config_data_logging_data[1]:
159                    if item in data:
160                        byte2 |= (1 << self.config_data_logging_data[1].index(item))
161
162                data_encoded = bytes([byte1, byte2])
163
164            else:
165                raise ValueError("Comando non supportato o dati non validi")
166        else:
167            # se il comando non richiede dati, usiamo un payload vuoto
168            data_encoded = bytes()
169

171        encoded_command = self.Header + self.dev_addr + instr_code + self._data_len(data_encoded) + data_encoded + self._checksum(instr_code, data_encoded)

257    def _continuous_read(self, data, interval=0.01):
258        """
259        Legge continuamente i dati dal buffer seriale, separando i byte e associando un timestamp a ciascuno, e li aggiunge a un buffer interno di lettura.
260        Questo metodo viene eseguito in un thread separato per permettere la lettura continua senza bloccare il thread principale.
261
262        Args:
263            data (list of str): I dati per la configurazione del data-logging, usati per eseguire l'handshake iniziale e assicurarsi che la bilancia stia inviando i dati desiderati.
264            interval (float, optional): L'intervallo di tempo tra ogni lettura. Defaults to 0.01.
265        """
266
267        # leggiamo l'intero buffer setiale e lo aggiungiamo al buffer interno di lettura, con il timestamp associato
268        while self.reading:
269            val = self.ser.read_all()
270            if val:
271                with self.lock:
272                    # separiamo i byte
273                    for byte in val:
274                        self._raw_read_buffer.append(byte)
275                        self._raw_timestamps.append(time.time())
276            
277            time.sleep(interval)

279    def _dummy_continuous_read(self, data, chunk_size=16, iterations=100):
280        """
281        Simula la lettura continua di dati, prendendo chunk di dati da una lista predefinita e aggiungendoli al buffer interno di lettura con un timestamp associato, per testare la logica di decodifica senza hardware reale.
282
283        Args:
284            data (list of bytes): I dati da simulare.
285            chunk_size (int, optional): La dimensione di ciascun chunk di dati. Defaults to 16.
286            iterations (int, optional): Il numero di iterazioni da eseguire. Defaults to 100.
287        """
288        i = 0
289        while self.reading and i < iterations * chunk_size:
290            val = data[i:i+chunk_size]  # Simula la lettura di chunk di dati
291            i = (i + chunk_size) % len(data)  # Loop attraverso i dati
292            with self.lock:
293                if val:
294                    for byte in val:
295                        self._raw_read_buffer.append(byte)
296                        self._raw_timestamps.append(time.time())
297            time.sleep(0.01)

299    def _decode_raw_buffer(self, sizes):
300        """
301        Decodifica i dati grezzi letti dal buffer seriale, cercando l'Header, estraendo i byte di indirizzo, codice di istruzione, lunghezza dei dati e i dati stessi, decodificando i dati in stringhe ASCII e aggiungendoli al buffer di lettura decodificato con il timestamp associato.
302
303        Args:
304            sizes (list of int): Le dimensioni dei chunk di dati da decodificare.
305        """
306        while len(self._raw_read_buffer) >= sum(sizes) + 2 + 3 + 1: # Header (2) + address+instr_code+data_length (3) + data + checksum (1)
307
308            # Controlla se i primi 2 byte sono l'Header
309            if self._raw_read_buffer[0] == 255 and self._raw_read_buffer[1] == 254:
310                # Rimuovi l'Header e i loro timestamp
311                self._raw_read_buffer.popleft()  # Rimuove 255
312                self._raw_read_buffer.popleft()  # Rimuove 254
313
314                self._raw_timestamps.popleft()  # Rimuove timestamp di 255
315                self._raw_timestamps.popleft()  # Rimuove timestamp di 254
316
317                # Rimuovi i byte di indirizzo, instr_code e data_length +  i timestamp associati (3 byte)
318                for _ in range(3):
319                    self._raw_read_buffer.popleft()
320                    self._raw_timestamps.popleft()
321
322                # Ora estrai i dati basati sui sizes specificati
323                data_bytes = []
324                for size in sizes:
325                    chunk = bytes([self._raw_read_buffer.popleft() for _ in range(size)])
326                    data_bytes.append(chunk)
327
328                # Decodifica i dati e aggiungili al buffer di lettura
329                decoded_data = [chunk.decode('ascii').strip() for chunk in data_bytes]
330                self.read_buffer.append(decoded_data)
331                self.timestamps.append(self._raw_timestamps.popleft())
332            else:
333                # Se non trovi l'Header, rimuovi il primo byte e continua a cercare
334                self._raw_read_buffer.popleft()
335                self._raw_timestamps.popleft()

449    def get_safe_reading(self):
450        # cerco l'Header [255, 254]
451        while True:
452            # leggo un byte alla volta finché non trovo l'inizio
453            b = self.ser.read(1)
454            if b == bytes([255]):
455                b2 = self.ser.read(1)
456                if b2 == bytes([254]):
457                    break
458
459        # leggi i 3 byte successivi: address, instr_code, length
460        header_info = self.ser.read(3)
461        addr = header_info[0]
462        instr_code = header_info[1]
463        instr_bytes = bytes([instr_code])
464        length = header_info[2]
465
466        # 3. leggi il corpo del messaggio (data) e il checksum finale
467        payload = self.ser.read(length)
468        received_chk = self.ser.read(1)[0]
469
470        # 4. verifica del checksum
471        calculated_chk_bytes = self.checksum(instr_bytes, payload) #checksum vuole dati in bytes
472        calculated_chk = calculated_chk_bytes[0]
473
474        if received_chk != calculated_chk:
475            print("Errore: Checksum non corrisponde! Pacchetto scartato.")
476            return None
477
478        # 5. Decodifica i dati (se sono ASCII)
479        
480        try:
481            valori_divisi = self.decode_ascii_data(payload, [length])
482            return valori_divisi 
483        except Exception as e:
484            print(f"Errore nella decodifica ASCII: {e}")
485            return None
